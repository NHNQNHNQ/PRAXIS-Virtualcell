```text

  --write-out 'default_proxy http_code=%{http_code} remote_ip=%{remote_ip} exit_marker\n' \
  http://10.102.106.53:7788/api/info

curl --noproxy '*' --max-time 5 --silent --show-error --output /dev/null \
  --write-out 'direct http_code=%{http_code} remote_ip=%{remote_ip} exit_marker\n' \
  http://10.102.106.53:7788/api/info

```
