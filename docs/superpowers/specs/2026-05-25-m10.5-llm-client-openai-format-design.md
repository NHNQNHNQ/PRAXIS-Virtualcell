## §3 OpenAI ↔ 内部 翻译细则

### 3.1 Outbound:caller messages(Anthropic 形)→ OpenAI 形

`query.py` 传给 `client.stream_messages()` 的 `messages` 是 Anthropic content-block 形:

```python
# Anthropic
{"role": "user", "content": "text only"}
{"role": "user", "content": [{"type": "tool_result", "tool_use_id": "T1", "content": "<result>"}]}
{"role": "assistant", "content": [{"type": "text", "text": "..."}]}
{"role": "assistant", "content": [
  {"type": "text", "text": "let me check"},
  {"type": "tool_use", "id": "T1", "name": "fs_read", "input": {"path": "/a"}},
]}
```

`client.py` 内部 `_translate_messages_anthropic_to_openai(messages, system)`:

```python
def _translate_messages_anthropic_to_openai(
    messages: list[dict], system: str | None,
) -> list[dict]:
    """Anthropic content-block messages → OpenAI flat messages。"""
    out: list[dict] = []
    if system:
        out.append({"role": "system", "content": system})
    for msg in messages:
        role = msg["role"]
        content = msg["content"]
        if isinstance(content, str):
            out.append({"role": role, "content": content})
            continue
        # content is list of blocks
        if role == "user":
            # tool_result blocks → role=tool messages(每个 tool_result 是独立 OpenAI message)
            for block in content:
                if block["type"] == "tool_result":
                    tool_content = block["content"]
                    if isinstance(tool_content, list):
                        # Anthropic tool_result content 可以是 [{type:"text", text:...}, ...]
                        tool_content = "".join(
                            b.get("text", "") for b in tool_content if isinstance(b, dict)
                        )
                    out.append({
                        "role": "tool",
                        "tool_call_id": block["tool_use_id"],
                        "content": str(tool_content),
                    })
                elif block["type"] == "text":
                    out.append({"role": "user", "content": block["text"]})
        elif role == "assistant":
            # 收集 text 和 tool_use,合并为单个 OpenAI assistant message
            texts: list[str] = []
            tool_calls: list[dict] = []
            for block in content:
                if block["type"] == "text":
                    texts.append(block["text"])
                elif block["type"] == "tool_use":
                    tool_calls.append({
                        "id": block["id"],
                        "type": "function",
                        "function": {
                            "name": block["name"],
                            "arguments": json.dumps(block["input"], ensure_ascii=False),
                        },
                    })
            assistant_msg: dict = {"role": "assistant"}
            if texts:
                assistant_msg["content"] = "".join(texts)
            else:
                assistant_msg["content"] = None   # OpenAI 允许 null content + tool_calls
            if tool_calls:
                assistant_msg["tool_calls"] = tool_calls
            out.append(assistant_msg)
    return out
```

### 3.3 Inbound:OpenAI SSE → 4 类规范事件

OpenAI 流式 chunk shape:

```json
data: {"id":"...", "object":"chat.completion.chunk", "choices":[{"index":0, "delta":{"role":"assistant", "content":"Hello"}, "finish_reason":null}], "usage": null}

data: {"choices":[{"delta":{"content":" world"}, "finish_reason":null}]}

data: {"choices":[{"delta":{"tool_calls":[{"index":0, "id":"call_abc", "type":"function", "function":{"name":"fs_read", "arguments":"{\""}}]}, "finish_reason":null}]}

data: {"choices":[{"delta":{"tool_calls":[{"index":0, "function":{"arguments":"path\":"}}]}, "finish_reason":null}]}

data: {"choices":[{"delta":{"tool_calls":[{"index":0, "function":{"arguments":"\"/etc\"}"}}]}, "finish_reason":null}]}

data: {"choices":[{"delta":{}, "finish_reason":"tool_calls"}], "usage":{"prompt_tokens":100,"completion_tokens":20}}

data: [DONE]
```

`parse_stream_response(http_response) -> AsyncIterator[dict]`:

```python
async def parse_stream_response(resp: httpx.Response) -> AsyncIterator[dict]:
    """OpenAI /v1/chat/completions stream → 4 类规范事件。"""
    # 累积器:tool_call index → {id, name, arguments_buf}
    tool_call_buf: dict[int, dict] = {}
    last_usage: dict = {}
    async for line in resp.aiter_lines():
        line = line.strip()
        if not line or not line.startswith("data:"):
            continue
        payload = line[5:].strip()
        if payload == "[DONE]":
            break
        try:
            ev = json.loads(payload)
        except json.JSONDecodeError:
            continue   # 跳过 malformed line
        # error 形 {"error": {...}}
        if "error" in ev:
            raise RuntimeError(f"LLM upstream error: {ev['error']}")
        if "usage" in ev and ev["usage"]:
            last_usage = ev["usage"]
        choices = ev.get("choices") or []
        if not choices:
            continue
        choice = choices[0]
        delta = choice.get("delta") or {}
        finish_reason = choice.get("finish_reason")

        # 文本流
        if delta.get("content"):
            yield {"type": "text", "text": delta["content"]}

        # thinking / reasoning_content 流
        if delta.get("reasoning_content"):
            yield {"type": "thinking", "text": delta["reasoning_content"]}

        # tool_calls 流(可能 partial)
        for tc in delta.get("tool_calls") or []:
            idx = tc.get("index", 0)
            buf = tool_call_buf.setdefault(idx, {
                "id": "", "name": "", "arguments_buf": "",
            })
            if tc.get("id"):
                buf["id"] = tc["id"]
            fn = tc.get("function") or {}
            if fn.get("name"):
                buf["name"] = fn["name"]
            if fn.get("arguments"):
                buf["arguments_buf"] += fn["arguments"]

        # finish_reason 出现时:flush 累积的 tool_calls + emit message_stop
        if finish_reason is not None:
            for idx in sorted(tool_call_buf):
                buf = tool_call_buf[idx]
                try:
                    args_dict = json.loads(buf["arguments_buf"]) if buf["arguments_buf"] else {}
                except json.JSONDecodeError:
                    args_dict = {"__raw__": buf["arguments_buf"]}
                yield {
                    "type": "tool_use",
                    "id": buf["id"],
                    "name": buf["name"],
                    "input": args_dict,
                }
            tool_call_buf.clear()
            # finish_reason 翻译
            stop_reason_map = {
                "stop": "end_turn",
                "tool_calls": "tool_use",
                "length": "max_tokens",
                "content_filter": "stop",
            }
            yield {
                "type": "message_stop",
                "stop_reason": stop_reason_map.get(finish_reason, finish_reason),
                "usage": last_usage,
            }
```

## §5 错误分类扩展

`praxis2/llm/retry.py:classify_withheld_error`:

```python
def classify_withheld_error(exc: Exception) -> WithheldKind:
    msg = str(exc).lower()

    # Anthropic 兜底(yaml 切净后实际不跑,留作 dead path 防回退)
    if "prompt is too long" in msg:
        return WithheldKind.PROMPT_TOO_LONG

    # OpenAI 错误码识别(M10.5 新加)
    # litellm 错误体形:{"error": {"message": "...", "code": "context_length_exceeded"}}
    if "context_length_exceeded" in msg or "context length exceeded" in msg:
        return WithheldKind.PROMPT_TOO_LONG
    if "rate_limit_exceeded" in msg or "rate limit" in msg or "429" in msg:
        return WithheldKind.RATE_LIMITED
    if "invalid_request_error" in msg or "invalid model name" in msg:
        return WithheldKind.UNKNOWN   # 不 recovery, query.py 出 ERROR

    return WithheldKind.UNKNOWN
```

`query.py:122-128` 的 recovery 分支不动(只看 `WithheldKind.PROMPT_TOO_LONG`)— OpenAI context_length_exceeded 触发 `await compactor.compact(state)` 正常 work,跟 Anthropic 路径同效。
