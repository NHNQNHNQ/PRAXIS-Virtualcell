# M11.1 Spawn 后台 LLM 任务 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 backend 加 `/api/spawn/*` 4 个真接口,把异步 LLM 任务跑通(创建/列表/查询/取消),持久化到 PG,完成时扣费,FE 现有 SpawnList 直接对接。

**Architecture:** 同进程 `asyncio.create_task` 跑 worker,`SpawnRegistry`(进程内 `dict[task_id, asyncio.Task]`)管 cancel。Worker 用现有 `LLMClient.stream_messages`(无 tool loop,one-shot)累积 text,每 20 chunk flush partial 到 PG `spawn_tasks.answer`。完成时一次性 `Ledger.record(-cost_li)`。每 user 最多 2 并发(active=pending+running),超额拒;每 user 保留近 100 条,insert 时 trim。

**Tech Stack:** Python 3.13 / asyncpg / Alembic Python migrations / raw ASGI(无 FastAPI)/ pytest-asyncio / `tests/_llm_fakes.FakeLLMClient` 复用。

```python
# praxis2/spawn/pricing.py
"""LLM 调用费用计算 — input/output token × 单价(li per 1M tokens)。"""
import math


def compute_cost_li(
    tokens_in: int, tokens_out: int,
    price_in_per_m: int, price_out_per_m: int,
) -> int:
    """tokens × 单价 / 1M, ceil 到整数 li。不足 1 li 也扣 1。"""
    if tokens_in <= 0 and tokens_out <= 0:
        return 0
    cost_micro = tokens_in * price_in_per_m + tokens_out * price_out_per_m
    return math.ceil(cost_micro / 1_000_000)
```

```python
# praxis2/spawn/repo.py
"""spawn_tasks 表 CRUD。

写路径:create / update_* 需调用方传 conn(便于跟 ledger 同事务);
读路径:get / list_recent / count_active 自取连接。
"""
from __future__ import annotations

import uuid
from typing import Any

import asyncpg


class SpawnRepo:
    def __init__(self, pool: asyncpg.Pool) -> None:
        self._pool = pool

    async def create(
        self, conn: asyncpg.Connection,
        user_id: str, prompt: str, model: str,
    ) -> str:
        uid = uuid.UUID(user_id) if isinstance(user_id, str) else user_id
        row = await conn.fetchrow(
            "INSERT INTO spawn_tasks (user_id, prompt, model) "
            "VALUES ($1, $2, $3) RETURNING id",
            uid, prompt, model,
        )
        return str(row["id"])

    async def get(self, task_id: str) -> dict[str, Any] | None:
        tid = uuid.UUID(task_id)
        async with self._pool.acquire() as conn:
            row = await conn.fetchrow(
                "SELECT id, user_id, prompt, model, status, answer, "
                "       tokens_in, tokens_out, cost_li, error, "
                "       created_at, started_at, completed_at "
                "FROM spawn_tasks WHERE id=$1", tid,
            )
        return _row_to_dict(row) if row else None

    async def list_recent(self, user_id: str, limit: int = 100) -> list[dict]:
        uid = uuid.UUID(user_id) if isinstance(user_id, str) else user_id
        async with self._pool.acquire() as conn:
            rows = await conn.fetch(
                "SELECT id, user_id, prompt, model, status, answer, "
                "       tokens_in, tokens_out, cost_li, error, "
                "       created_at, started_at, completed_at "
                "FROM spawn_tasks WHERE user_id=$1 "
                "ORDER BY created_at DESC LIMIT $2",
                uid, int(limit),
            )
        return [_row_to_dict(r) for r in rows]

    async def count_active(self, user_id: str) -> int:
        uid = uuid.UUID(user_id) if isinstance(user_id, str) else user_id
        async with self._pool.acquire() as conn:
            return await conn.fetchval(
                "SELECT count(*) FROM spawn_tasks "
                "WHERE user_id=$1 AND status IN ('pending','running')",
                uid,
            )

    async def update_running(self, task_id: str) -> None:
        tid = uuid.UUID(task_id)
        async with self._pool.acquire() as conn:
            await conn.execute(
                "UPDATE spawn_tasks SET status='running', started_at=now() "
                "WHERE id=$1", tid,
            )

    async def update_partial(self, task_id: str, answer_so_far: str) -> None:
        tid = uuid.UUID(task_id)
        async with self._pool.acquire() as conn:
            await conn.execute(
                "UPDATE spawn_tasks SET answer=$2 WHERE id=$1",
                tid, answer_so_far,
            )

    async def update_done(
        self, conn: asyncpg.Connection,
        task_id: str, answer: str,
        tokens_in: int, tokens_out: int, cost_li: int,
    ) -> None:
        tid = uuid.UUID(task_id)
        await conn.execute(
            "UPDATE spawn_tasks SET status='done', answer=$2, "
            "tokens_in=$3, tokens_out=$4, cost_li=$5, completed_at=now() "
            "WHERE id=$1",
            tid, answer, int(tokens_in), int(tokens_out), int(cost_li),
        )

    async def update_failed(self, task_id: str, error_msg: str) -> None:
        tid = uuid.UUID(task_id)
        async with self._pool.acquire() as conn:
            await conn.execute(
                "UPDATE spawn_tasks SET status='failed', error=$2, "
                "completed_at=now() WHERE id=$1",
                tid, error_msg[:2000],
            )

    async def update_cancelled(self, task_id: str) -> None:
        tid = uuid.UUID(task_id)
        async with self._pool.acquire() as conn:
            await conn.execute(
                "UPDATE spawn_tasks SET status='cancelled', completed_at=now() "
                "WHERE id=$1", tid,
            )

    async def trim_old(self, user_id: str, keep: int = 100) -> int:
        """删超出 keep 条之外的老任务。返删除行数。"""
        uid = uuid.UUID(user_id) if isinstance(user_id, str) else user_id
        async with self._pool.acquire() as conn:
            result = await conn.execute(
                "DELETE FROM spawn_tasks WHERE id IN ("
                "  SELECT id FROM spawn_tasks WHERE user_id=$1 "
                "  ORDER BY created_at DESC OFFSET $2"
                ")",
                uid, int(keep),
            )
        # asyncpg execute 返 "DELETE N"
        return int(result.split()[-1]) if result else 0


def _row_to_dict(r) -> dict[str, Any]:
    return {
        "id": str(r["id"]),
        "user_id": str(r["user_id"]),
        "prompt": r["prompt"],
        "model": r["model"],
        "status": r["status"],
        "answer": r["answer"],
        "tokens_in": int(r["tokens_in"]),
        "tokens_out": int(r["tokens_out"]),
        "cost_li": int(r["cost_li"]),
        "error": r["error"],
        "created_at": r["created_at"],
        "started_at": r["started_at"],
        "completed_at": r["completed_at"],
    }
```

```python
stale running → failed。
"""
from __future__ import annotations

import asyncio


class SpawnRegistry:
    def __init__(self) -> None:
        self._tasks: dict[str, asyncio.Task] = {}

    def register(self, task_id: str, task: asyncio.Task) -> None:
        self._tasks[task_id] = task

    def cancel(self, task_id: str) -> bool:
        """cancel 指定 task。返 True=找到并发起 cancel,False=没找到/已完成。"""
        task = self._tasks.get(task_id)
        if task is None or task.done():
            return False
        task.cancel()
        return True

    def unregister(self, task_id: str) -> None:
        self._tasks.pop(task_id, None)

    def active_count(self) -> int:
        return sum(1 for t in self._tasks.values() if not t.done())
```

```python
# praxis2/spawn/worker.py
"""异步执行单个 spawn task — 拉 LLM stream, 累积 text, 完成时扣费。

不走 query.py 完整 loop:spawn 是 one-shot 无 tool 调用。
错误处理 (Q8):任何异常 → status=failed + error 存。
取消:asyncio.CancelledError → status=cancelled, 不扣费(Q5)。
"""
from __future__ import annotations

import logging

from praxis2.accounting.ledger import Ledger
from praxis2.spawn.pricing import compute_cost_li
from praxis2.spawn.repo import SpawnRepo

log = logging.getLogger(__name__)


async def run_spawn_task(
    *,
    task_id: str,
    user_id: str,
    prompt: str,
    model: str,
    client,                       # LLMClient or FakeLLMClient (鸭子类型)
    repo: SpawnRepo,
    ledger: Ledger,
    pg_pool,                      # asyncpg.Pool — 显式传,不取 ledger._pool
    max_tokens: int,
    price_in_per_m: int,
    price_out_per_m: int,
    partial_flush_chunks: int,
    keep_last_per_user: int,
) -> None:
    """单条 spawn 任务执行体。被 asyncio.create_task 包,SpawnRegistry 管 cancel。

    不返结果(状态写 PG;handler poll 拿)。不抛(全捕,落库为 failed/cancelled)。
    """
    import asyncio
    await repo.update_running(task_id)

    chunks: list[str] = []
    flush_counter = 0
    tokens_in = 0
    tokens_out = 0
    try:
        messages = [{"role": "user", "content": prompt}]
        async for ev in client.stream_messages(
            model=model, messages=messages, max_tokens=max_tokens,
        ):
            etype = ev.get("type")
            if etype == "text":
                chunks.append(ev.get("text", ""))
                flush_counter += 1
                if flush_counter >= partial_flush_chunks:
                    await repo.update_partial(task_id, "".join(chunks))
                    flush_counter = 0
            elif etype == "message_stop":
                usage = ev.get("usage") or {}
                tokens_in = int(usage.get("input_tokens", 0) or 0)
                tokens_out = int(usage.get("output_tokens", 0) or 0)
                break

        final_answer = "".join(chunks)
        cost = compute_cost_li(
            tokens_in, tokens_out, price_in_per_m, price_out_per_m,
        )

        async with pg_pool.acquire() as conn:
            async with conn.transaction():
                await repo.update_done(
                    conn, task_id, final_answer, tokens_in, tokens_out, cost,
                )
                if cost > 0:
                    await ledger.record(
                        conn, user_id, -cost,
                        kind="spawn_consume",
                        ref_type="spawn_task", ref_id=task_id,
                        meta={"tokens_in": tokens_in, "tokens_out": tokens_out},
                    )

        # TTL trim (Q7) — 同步执行,无需等
        await repo.trim_old(user_id, keep=keep_last_per_user)

    except asyncio.CancelledError:
        await repo.update_cancelled(task_id)
        raise
    except Exception as exc:
        log.exception("spawn task %s failed", task_id)
        await repo.update_failed(task_id, repr(exc))
```

```python
# praxis2/spawn/handlers.py
"""4 个 spawn ASGI handler。

挂在 server/app.py:
- POST /api/spawn                      handle_spawn_post
- GET  /api/spawn/list                 handle_spawn_list
- GET  /api/spawn/get/{task_id}        handle_spawn_get
- POST /api/spawn/cancel/{task_id}     handle_spawn_cancel
"""
from __future__ import annotations

import asyncio
import json
from typing import Any


async def _json(send, status: int, body: dict) -> None:
    data = json.dumps(body, ensure_ascii=False, default=str).encode()
    await send({"type": "http.response.start", "status": status,
                "headers": [(b"content-type", b"application/json; charset=utf-8")]})
    await send({"type": "http.response.body", "body": data})


async def _read_body(receive) -> dict:
    chunks = b""
    while True:
        msg = await receive()
        chunks += msg.get("body", b"")
        if not msg.get("more_body"):
            break
    if not chunks:
        return {}
    try:
        return json.loads(chunks)
    except ValueError:
        return {}


def _require_user(scope) -> tuple[str, Any]:
    user = scope.get("user")
    if not user:
        return ("unauthorized", None)
    return ("ok", user)


async def handle_spawn_post(scope, receive, send, deps) -> None:
    """POST /api/spawn body {prompt, model?} → 创任务 + 启 asyncio 跑 worker。"""
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return

    body = await _read_body(receive)
    prompt = (body.get("prompt") or "").strip()
    if not prompt:
        await _json(send, 400, {"error": "missing_prompt"})
        return
    model = body.get("model") or deps["default_model"]
    if not isinstance(model, str) or not model:
        await _json(send, 400, {"error": "bad_model"})
        return

    user_id = user["id"]
    repo = deps["spawn_repo"]
    registry = deps["spawn_registry"]
    ledger = deps["ledger"]

    # concurrent 限
    active = await repo.count_active(user_id)
    if active >= deps["max_concurrent_per_user"]:
        await _json(send, 429, {"error": "too_many_active",
                                 "active": active,
                                 "limit": deps["max_concurrent_per_user"]})
        return

    # quota 预检 — 完成时才真扣,但 balance 必须 > 0
    async with deps["pg_pool"].acquire() as conn:
        bal = await conn.fetchval(
            "SELECT balance_li FROM users WHERE id=$1", user_id,
        )
    if bal is None or bal <= 0:
        await _json(send, 402, {"error": "insufficient_balance", "balance": bal})
        return

    # 创 row
    async with deps["pg_pool"].acquire() as conn:
        task_id = await repo.create(conn, user_id, prompt, model)

    # 启 asyncio task
    from praxis2.spawn.worker import run_spawn_task
    coro = run_spawn_task(
        task_id=task_id, user_id=user_id, prompt=prompt, model=model,
        client=deps["llm_client_factory"](),
        repo=repo, ledger=ledger, pg_pool=deps["pg_pool"],
        max_tokens=deps["max_tokens"],
        price_in_per_m=deps["price_in_per_m"],
        price_out_per_m=deps["price_out_per_m"],
        partial_flush_chunks=deps["partial_flush_chunks"],
        keep_last_per_user=deps["keep_last_per_user"],
    )
    task = asyncio.create_task(coro)
    registry.register(task_id, task)
    task.add_done_callback(lambda _t: registry.unregister(task_id))

    await _json(send, 200, {"id": task_id, "status": "pending"})


async def handle_spawn_list(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return
    rows = await deps["spawn_repo"].list_recent(user["id"], limit=100)
    await _json(send, 200, {"tasks": rows})


async def handle_spawn_get(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return
    task_id = scope.get("path_params", {}).get("task_id", "")
    if not task_id:
        await _json(send, 400, {"error": "missing_task_id"})
        return
    row = await deps["spawn_repo"].get(task_id)
    if row is None:
        await _json(send, 404, {"error": "not_found"})
        return
    if row["user_id"] != user["id"] and user.get("role") != "admin":
        await _json(send, 404, {"error": "not_found"})   # 不泄露存在
        return
    await _json(send, 200, row)


async def handle_spawn_cancel(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return
    task_id = scope.get("path_params", {}).get("task_id", "")
    if not task_id:
        await _json(send, 400, {"error": "missing_task_id"})
        return
    row = await deps["spawn_repo"].get(task_id)
    if row is None or (row["user_id"] != user["id"] and user.get("role") != "admin"):
        await _json(send, 404, {"error": "not_found"})
        return
    if row["status"] not in ("pending", "running"):
        await _json(send, 409, {"error": "not_active", "status": row["status"]})
        return
    cancelled = deps["spawn_registry"].cancel(task_id)
    # 即使 registry 没找到(进程跨重启),仍把 DB 标 cancelled
    if not cancelled:
        await deps["spawn_repo"].update_cancelled(task_id)
    await _json(send, 200, {"ok": True})
```
