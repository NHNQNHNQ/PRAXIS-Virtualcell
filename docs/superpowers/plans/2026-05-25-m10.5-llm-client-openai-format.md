def classify_withheld_error(exc: Exception) -> WithheldKind:
    msg = str(exc).lower()

    # Anthropic 兜底(yaml 切净后实际不跑,留作 dead path 防回退)
    if "prompt is too long" in msg:
        return WithheldKind.PROMPT_TOO_LONG

    # M10.5:OpenAI error code 识别
    # litellm 错误体形:{"error":{"message":"...", "code":"context_length_exceeded"}}
    if "context_length_exceeded" in msg or "context length exceeded" in msg:
        return WithheldKind.PROMPT_TOO_LONG
    if "rate_limit_exceeded" in msg or "rate limit" in msg or "429" in msg:
        return WithheldKind.RATE_LIMITED
    if "invalid_request_error" in msg or "invalid model name" in msg:
        return WithheldKind.UNKNOWN

    return WithheldKind.UNKNOWN
```

错误:upstream 200 + body 含 'error' 字段 → 抛 RuntimeError;
HTTP 非 2xx → caller 应在 raise_for_status 抛 httpx.HTTPStatusError 之前调用此函数,

"""

from __future__ import annotations

import json
from collections.abc import AsyncIterator
from typing import Any

import httpx

# OpenAI finish_reason → 内部 stop_reason
_STOP_REASON_MAP = {
    "stop": "end_turn",
    "tool_calls": "tool_use",
    "length": "max_tokens",
    "content_filter": "stop",
}


async def parse_stream_response(
    resp: httpx.Response,
) -> AsyncIterator[dict[str, Any]]:
    """OpenAI /v1/chat/completions stream → 4 类规范事件。"""
    tool_call_buf: dict[int, dict[str, Any]] = {}
    last_usage: dict[str, Any] = {}

    async for line in resp.aiter_lines():
        line = line.strip()
        if not line:
            continue
        if not line.startswith("data:"):
            continue
        payload = line[len("data:"):].strip()
        if payload == "[DONE]":
            break
        try:
            ev = json.loads(payload)
        except json.JSONDecodeError:
            continue   # 跳过 malformed line

        # 错误体(litellm 偶尔以 200 + {"error":...} 返)
        if isinstance(ev, dict) and "error" in ev:
            err = ev["error"]
            msg = err.get("message", str(err)) if isinstance(err, dict) else str(err)
            raise RuntimeError(f"LLM upstream error: {msg}")

        # usage 可能在最后一帧 或 stream_options=include_usage 中早出
        if ev.get("usage"):
            last_usage = ev["usage"]

        choices = ev.get("choices") or []
        if not choices:
            continue
        choice = choices[0]
        delta = choice.get("delta") or {}
        finish_reason = choice.get("finish_reason")

        # 文本流
        content = delta.get("content")
        if content:
            yield {"type": "text", "text": content}

        # reasoning / thinking
        reasoning = delta.get("reasoning_content")
        if reasoning:
            yield {"type": "thinking", "text": reasoning}

        # tool_calls 流(partial JSON args 累积)
        for tc in delta.get("tool_calls") or []:
            idx = tc.get("index", 0)
            buf = tool_call_buf.setdefault(idx, {
                "id": "", "name": "", "arguments_buf": "",
            })
            if tc.get("id"):
                buf["id"] = tc["id"]
            fn = tc.get("function") or {}
            if fn.get("name"):
                buf["name"] = fn["name"]
            if fn.get("arguments"):
                buf["arguments_buf"] += fn["arguments"]

        # finish_reason 出现:flush 累积 tool_calls + emit message_stop
        if finish_reason is not None:
            for idx in sorted(tool_call_buf):
                buf = tool_call_buf[idx]
                try:
                    args_dict = (
                        json.loads(buf["arguments_buf"])
                        if buf["arguments_buf"]
                        else {}
                    )
                except json.JSONDecodeError:
                    args_dict = {"__raw__": buf["arguments_buf"]}
                yield {
                    "type": "tool_use",
                    "id": buf["id"],
                    "name": buf["name"],
                    "input": args_dict,
                }
            tool_call_buf.clear()
            yield {
                "type": "message_stop",
                "stop_reason": _STOP_REASON_MAP.get(finish_reason, finish_reason),
                "usage": last_usage,
            }
```

"""LLM 客户端(M10.5:OpenAI /v1/chat/completions 协议)。

旧 Anthropic /v1/messages 路径已删。litellm proxy 提供 OpenAI-compatible
endpoint,真路由到 deepseek-v4-pro / deepseek-v4-flash。

backward alias `AnthropicClient = LLMClient` 保留(`server/app.py` + 老测试仍
import 老名)。

from __future__ import annotations

import json
from typing import Any

import httpx

from praxis2.llm.sse import parse_stream_response


class LLMClient:
    def __init__(
        self,
        api_key: str,
        base_url: str,
        timeout_s: float = 60.0,
    ) -> None:
        self._headers = {
            "Authorization": f"Bearer {api_key}",
        }
        self._http = httpx.AsyncClient(
            base_url=base_url.rstrip("/"),
            timeout=timeout_s,
        )

    async def messages(
        self,
        *,
        model: str,
        messages: list[dict[str, Any]],
        max_tokens: int,
        system: str | list[dict] | None = None,
        tools: list[dict[str, Any]] | None = None,
    ) -> dict[str, Any]:
        """非流式 — M0 旧 anthropic 接口位保留(目前 query.py 不调,留作工具)。"""
        payload = self._build_payload(
            model=model, messages=messages, max_tokens=max_tokens,
            system=system, tools=tools, stream=False,
        )
        resp = await self._http.post(
            "/v1/chat/completions", headers=self._headers, json=payload,
        )
        resp.raise_for_status()
        return resp.json()

    async def stream_messages(
        self,
        *,
        model: str,
        messages: list[dict[str, Any]],
        max_tokens: int,
        system: str | list[dict] | None = None,
        tools: list[dict[str, Any]] | None = None,
    ):
        """流式 — query.py 调。yields 4 类规范事件。"""
        payload = self._build_payload(
            model=model, messages=messages, max_tokens=max_tokens,
            system=system, tools=tools, stream=True,
        )
        payload["stream_options"] = {"include_usage": True}
        async with self._http.stream(
            "POST", "/v1/chat/completions",
            headers=self._headers, json=payload,
        ) as resp:
            resp.raise_for_status()
            async for ev in parse_stream_response(resp):
                yield ev

    def _build_payload(
        self,
        *,
        model: str,
        messages: list[dict[str, Any]],
        max_tokens: int,
        system: str | list[dict] | None,
        tools: list[dict[str, Any]] | None,
        stream: bool,
    ) -> dict[str, Any]:
        openai_messages = _translate_messages_anthropic_to_openai(messages, system)
        payload: dict[str, Any] = {
            "model": model,
            "messages": openai_messages,
            "max_tokens": max_tokens,
        }
        if stream:
            payload["stream"] = True
        if tools:
            payload["tools"] = tools
        return payload

    async def aclose(self) -> None:
        await self._http.aclose()


# Backward alias(server/app.py + tests 仍 import 老名)
AnthropicClient = LLMClient


def _translate_messages_anthropic_to_openai(
    messages: list[dict[str, Any]],
    system: str | list[dict] | None,
) -> list[dict[str, Any]]:
    """Anthropic content-block messages → OpenAI flat messages。

    - system top-level → messages[0] role=system
    - {role:'user', content:'str'} → 不变
    - {role:'user', content:[{type:'tool_result',...}]} → role='tool' messages(每 tool_result 一条)
    - {role:'assistant', content:[{type:'text',...},{type:'tool_use',...}]} → 合并为 1 条 assistant message(content + tool_calls)
    """
    out: list[dict[str, Any]] = []

    if system:
        # system 可能是 str 或 list[dict](Anthropic 形);统一为 str
        if isinstance(system, list):
            sys_text = "".join(
                b.get("text", "") for b in system
                if isinstance(b, dict) and b.get("type") == "text"
            )
        else:
            sys_text = system
        out.append({"role": "system", "content": sys_text})

    for msg in messages:
        role = msg["role"]
        content = msg["content"]

        # plain text content
        if isinstance(content, str):
            out.append({"role": role, "content": content})
            continue

        # content is list of blocks(Anthropic 形)
        if role == "user":
            # tool_result blocks → 每个 → role=tool message;text blocks → 合并为 user message
            user_texts: list[str] = []
            for block in content:
                btype = block.get("type")
                if btype == "tool_result":
                    tool_content = block.get("content", "")
                    if isinstance(tool_content, list):
                        tool_content = "".join(
                            b.get("text", "") for b in tool_content
                            if isinstance(b, dict)
                        )
                    out.append({
                        "role": "tool",
                        "tool_call_id": block["tool_use_id"],
                        "content": str(tool_content),
                    })
                elif btype == "text":
                    user_texts.append(block.get("text", ""))
            if user_texts:
                out.append({"role": "user", "content": "".join(user_texts)})

        elif role == "assistant":
            # 合并 text + tool_use 为单 assistant message
            texts: list[str] = []
            tool_calls: list[dict] = []
            for block in content:
                btype = block.get("type")
                if btype == "text":
                    texts.append(block.get("text", ""))
                elif btype == "tool_use":
                    tool_calls.append({
                        "id": block["id"],
                        "type": "function",
                        "function": {
                            "name": block["name"],
                            "arguments": json.dumps(
                                block.get("input", {}), ensure_ascii=False,
                            ),
                        },
                    })
            assistant_msg: dict[str, Any] = {"role": "assistant"}
            text_content = "".join(texts)
            # OpenAI 允许 content=null 当 tool_calls 存在
            assistant_msg["content"] = text_content if text_content else None
            if tool_calls:
                assistant_msg["tool_calls"] = tool_calls
            out.append(assistant_msg)

    return out
