"""MD 作业计费 — GPU 小时 × 单价 + CPU 溢额小时 × 单价,ceil 到整数 li。"""
import math


def compute_md_cost_li(
    gpu_hours: float, cpu_hours: float,
    free_cpu_hours: float,
    price_gpu_per_h_li: int, price_cpu_per_h_li: int,
) -> int:
    """gpu_h × p_gpu + max(0, cpu_h - free) × p_cpu, ceil 到 li。"""
    cpu_overage = max(0.0, cpu_hours - free_cpu_hours)
    raw = gpu_hours * price_gpu_per_h_li + cpu_overage * price_cpu_per_h_li
    return math.ceil(raw)
```

# praxis2/md/sim_params.py
"""3 sim_type 必填字段校验 — 返 err str(handler 转 400),None=OK。"""
from __future__ import annotations


class SimParamsError(Exception):
    pass


# (field_name, type 校验函数)
_SMD_REQUIRED: list[tuple[str, type]] = [
    ("pull_coord_geometry", str),
    ("pull_rate", (int, float)),
    ("pull_k1", (int, float)),
    ("pull_group1_name", str),
    ("pull_group2_name", str),
]

_META_REQUIRED: list[tuple[str, type]] = [
    ("plumed_file_id", str),
]

_SUPPORTED = {"standard", "smd_pulling", "metadynamics"}


def validate_sim_params(sim_type: str, sim_params: dict) -> str | None:
    """返 None=OK,否则返错误字符串(supplied to handler 转 400)。"""
    if sim_type not in _SUPPORTED:
        return "unsupported_sim_type"
    if sim_type == "standard":
        return None
    if sim_type == "smd_pulling":
        return _check_required(sim_params, _SMD_REQUIRED)
    if sim_type == "metadynamics":
        return _check_required(sim_params, _META_REQUIRED)
    return None


def _check_required(
    params: dict, required: list[tuple[str, type | tuple]],
) -> str | None:
    for name, expect_type in required:
        if name not in params:
            return f"missing: {name}"
        v = params[name]
        if not isinstance(v, expect_type):
            return f"type: {name}"
    return None
```

# praxis2/md/sbatch.py
"""3 个 inline sbatch 模板 — standard / smd_pulling / metadynamics。

每个模板都是完整的 bash 脚本,包含:
- SBATCH header (account / nodes / gpus / time)
- GROMACS grompp + mdrun 命令
- exit code 透传(SLURM 经 sacct 拿)
"""
from __future__ import annotations

_BASE_HEADER = """#!/bin/bash
#SBATCH --account={account}
#SBATCH --nodes=1
#SBATCH --ntasks=1
#SBATCH --gres=gpu:1
#SBATCH --time=24:00:00
#SBATCH --job-name=praxis_md_{sim_type}
#SBATCH --output={work_dir}/slurm.out
#SBATCH --error={work_dir}/slurm.err

set -euo pipefail
cd {work_dir}

# GROMACS 加载 (节点环境)
module load gromacs 2>/dev/null || source /opt/gromacs/bin/GMXRC

# grompp:input.pdb + force_field/water_model → topol.top
echo q | gmx pdb2gmx -f input.pdb -o conf.gro -p topol.top \\
    -ff {force_field} -water {water_model}

# 写 mdp(NVT/NPT 含温度 + dt + nsteps)
NSTEPS=$(python3 -c "print(int({time_ns} * 1000 / {timestep_fs}))")
cat > md.mdp <<MDP
integrator = md
dt         = {timestep_fs_ps}
nsteps     = $NSTEPS
nstxout    = 5000
nstvout    = 5000
nstenergy  = 5000
nstlog     = 5000
{ensemble_block}
ref_t      = {temperature_k}
{extra_mdp}
MDP

gmx grompp -f md.mdp -c conf.gro -p topol.top -o md.tpr -maxwarn 2

# mdrun (gpu)
gmx mdrun -deffnm md -nb gpu {extra_mdrun_flags}

echo "MD finished"
exit 0
"""

_ENSEMBLE_NVT = "tcoupl     = v-rescale\ntc-grps    = System"
_ENSEMBLE_NPT = (
    "tcoupl     = v-rescale\n"
    "tc-grps    = System\n"
    "pcoupl     = c-rescale\n"
    "ref_p      = 1.0\n"
    "compressibility = 4.5e-5"
)


def render_sbatch(
    *,
    sim_type: str,
    params: dict,
    sim_params: dict,
    work_dir: str,
    account: str,
) -> str:
    """生成完整 sbatch 脚本。"""
    ensemble_block = (
        _ENSEMBLE_NPT if params["ensemble"].upper() == "NPT" else _ENSEMBLE_NVT
    )
    extra_mdp = _build_extra_mdp(sim_type, sim_params)
    extra_mdrun_flags = _build_extra_mdrun(sim_type, sim_params)
    return _BASE_HEADER.format(
        account=account,
        sim_type=sim_type,
        work_dir=work_dir,
        force_field=params["force_field"],
        water_model=params["water_model"],
        timestep_fs=params["timestep_fs"],
        timestep_fs_ps=params["timestep_fs"] / 1000.0,
        time_ns=params["time_ns"],
        ensemble_block=ensemble_block,
        ref_t=params["temperature_k"],
        temperature_k=params["temperature_k"],
        extra_mdp=extra_mdp,
        extra_mdrun_flags=extra_mdrun_flags,
    )


def _build_extra_mdp(sim_type: str, sim_params: dict) -> str:
    if sim_type == "smd_pulling":
        return (
            "pull = yes\n"
            "pull_ncoords = 1\n"
            "pull_ngroups = 2\n"
            f"pull_group1_name = {sim_params['pull_group1_name']}\n"
            f"pull_group2_name = {sim_params['pull_group2_name']}\n"
            "pull_coord1_groups = 1 2\n"
            f"pull_coord1_geometry = {sim_params['pull_coord_geometry']}\n"
            "pull_coord1_type = umbrella\n"
            f"pull_coord1_rate = {sim_params['pull_rate']}\n"
            f"pull_coord1_k = {sim_params['pull_k1']}"
        )
    return ""


def _build_extra_mdrun(sim_type: str, sim_params: dict) -> str:
    if sim_type == "metadynamics":
        return "-plumed input.plumed"
    return ""
```

# praxis2/md/ssh.py
"""asyncssh 封装 — 每次新 connect / async with close。

不维护 connection pool — 简化(M11.2 carry 提到优化点)。
ssh-key 路径取 SshConfig.key_path,known_hosts 用宽松 (StrictHostKeyChecking=no
等价)避免运维负担。
"""
from __future__ import annotations

from dataclasses import dataclass

import asyncssh


@dataclass(frozen=True)
class SshConfig:
    host: str
    user: str
    key_path: str
    timeout_s: float = 15.0


async def _connect(cfg: SshConfig):
    """每次新建 connect — caller 用 async with cm。"""
    return await asyncssh.connect(
        host=cfg.host,
        username=cfg.user,
        client_keys=[cfg.key_path],
        known_hosts=None,                  # 宽松,M11.2 carry 加 host pinning
        connect_timeout=cfg.timeout_s,
    )


async def ssh_run(
    cfg: SshConfig, cmd: str, *, stdin: bytes | None = None,
) -> str:
    """跑 cmd, 返 stdout. 非 0 exit code 抛 RuntimeError(含 stderr)。"""
    async with await _connect(cfg) as conn:
        result = await conn.run(cmd, input=stdin, check=False)
    if result.returncode != 0:
        raise RuntimeError(
            f"ssh_failed cmd={cmd!r} rc={result.returncode} "
            f"stderr={(result.stderr or '').strip()[:500]}"
        )
    return result.stdout or ""


async def ssh_write_file(
    cfg: SshConfig, remote_path: str, content: bytes,
) -> None:
    """SFTP 写文件到 remote_path. 覆盖已存在。"""
    async with await _connect(cfg) as conn:
        async with conn.start_sftp_client() as sftp:
            async with sftp.open(remote_path, "wb") as f:
                await f.write(content)
```

# praxis2/md/repo.py
"""md_jobs 表 CRUD + finalize 接 ledger 同事务。"""
from __future__ import annotations

import json
import uuid
from typing import Any

import asyncpg


class MdJobRepo:
    def __init__(self, pool: asyncpg.Pool) -> None:
        self._pool = pool

    async def create(
        self, conn: asyncpg.Connection,
        user_id: str, sim_type: str, params: dict,
    ) -> str:
        uid = uuid.UUID(user_id) if isinstance(user_id, str) else user_id
        row = await conn.fetchrow(
            "INSERT INTO md_jobs (user_id, sim_type, params) "
            "VALUES ($1, $2, $3::jsonb) RETURNING id",
            uid, sim_type, json.dumps(params),
        )
        return str(row["id"])

    async def get(self, job_id: str) -> dict[str, Any] | None:
        jid = uuid.UUID(job_id)
        async with self._pool.acquire() as conn:
            row = await conn.fetchrow(_SELECT + " WHERE id=$1", jid)
        return _row_to_dict(row) if row else None

    async def list_recent(self, user_id: str, limit: int = 100) -> list[dict]:
        uid = uuid.UUID(user_id) if isinstance(user_id, str) else user_id
        async with self._pool.acquire() as conn:
            rows = await conn.fetch(
                _SELECT + " WHERE user_id=$1 ORDER BY created_at DESC LIMIT $2",
                uid, int(limit),
            )
        return [_row_to_dict(r) for r in rows]

    async def list_active(self) -> list[dict]:
        async with self._pool.acquire() as conn:
            rows = await conn.fetch(
                _SELECT + " WHERE status IN ('queued','running') "
                "ORDER BY created_at ASC",
            )
        return [_row_to_dict(r) for r in rows]

    async def count_active(self, user_id: str) -> int:
        uid = uuid.UUID(user_id) if isinstance(user_id, str) else user_id
        async with self._pool.acquire() as conn:
            return await conn.fetchval(
                "SELECT count(*) FROM md_jobs "
                "WHERE user_id=$1 AND status IN ('queued','running')",
                uid,
            )

    async def update_submitted(self, job_id: str, *, slurm_job_id: str) -> None:
        jid = uuid.UUID(job_id)
        async with self._pool.acquire() as conn:
            await conn.execute(
                "UPDATE md_jobs SET slurm_job_id=$2 WHERE id=$1",
                jid, slurm_job_id,
            )

    async def update_status(
        self, job_id: str, status: str, *, gpu_host: str | None = None,
    ) -> None:
        jid = uuid.UUID(job_id)
        async with self._pool.acquire() as conn:
            if status == "running":
                await conn.execute(
                    "UPDATE md_jobs SET status='running', started_at=now(), "
                    "gpu_host=COALESCE($2, gpu_host) "
                    "WHERE id=$1 AND status='queued'",
                    jid, gpu_host,
                )
            else:
                await conn.execute(
                    "UPDATE md_jobs SET status=$2 WHERE id=$1",
                    jid, status,
                )

    async def update_cancelled(self, job_id: str) -> None:
        jid = uuid.UUID(job_id)
        async with self._pool.acquire() as conn:
            await conn.execute(
                "UPDATE md_jobs SET status='cancelled', finished_at=now() "
                "WHERE id=$1 AND status IN ('queued','running')",
                jid,
            )

    async def finalize(
        self, conn: asyncpg.Connection,
        job_id: str, *, status: str,
        cpu_hours: float, gpu_hours: float, cost_li: int,
        error: str | None,
    ) -> None:
        """终态写入 — 必须在调用方事务里(ledger.record 同事务)。"""
        jid = uuid.UUID(job_id)
        await conn.execute(
            "UPDATE md_jobs SET status=$2, cpu_hours=$3, gpu_hours=$4, "
            "cost_li=$5, error=$6, finished_at=now() WHERE id=$1",
            jid, status, float(cpu_hours), float(gpu_hours),
            int(cost_li), (error or None),
        )


_SELECT = (
    "SELECT id, user_id, sim_type, params, status, slurm_job_id, "
    "       gpu_host, progress, error, cost_li, "
    "       cpu_hours, gpu_hours, "
    "       created_at, started_at, finished_at "
    "FROM md_jobs"
)


def _row_to_dict(r) -> dict[str, Any]:
    p = r["params"]
    if isinstance(p, str):
        p = json.loads(p)
    return {
        "id": str(r["id"]),
        "user_id": str(r["user_id"]),
        "sim_type": r["sim_type"],
        "params": p,
        "status": r["status"],
        "slurm_job_id": r["slurm_job_id"],
        "gpu_host": r["gpu_host"],
        "progress": int(r["progress"]),
        "error": r["error"],
        "cost_li": int(r["cost_li"]),
        "cpu_hours": float(r["cpu_hours"]),
        "gpu_hours": float(r["gpu_hours"]),
        "created_at": r["created_at"],
        "started_at": r["started_at"],
        "finished_at": r["finished_at"],
    }
```

# praxis2/md/poll_worker.py
"""后台 30s 轮询 worker — squeue scan active jobs + sacct 终态 + ledger 扣费。

启动:backend startup hook 起一个 asyncio.create_task(run_poll_loop(...))。
进程崩 → 丢 worker(已知 carry,M11.x 修)。
"""
from __future__ import annotations

import asyncio
import logging

from praxis2.accounting.ledger import Ledger
from praxis2.md.pricing import compute_md_cost_li
from praxis2.md.repo import MdJobRepo
from praxis2.md.ssh import SshConfig, ssh_run

log = logging.getLogger(__name__)


# sacct State → schema status
_SACCT_OK = {"COMPLETED"}
_SACCT_FAIL = {"FAILED", "NODE_FAIL", "OUT_OF_MEMORY", "BOOT_FAIL", "TIMEOUT"}
_SACCT_CANCEL = {"CANCELLED", "CANCELLED+"}


async def run_poll_loop(
    *, cfg: SshConfig, repo: MdJobRepo, ledger: Ledger, pg_pool,
    interval_s: int, free_cpu_h: float,
    price_gpu: int, price_cpu: int,
) -> None:
    """无限循环,每 interval_s 跑一次 poll_tick。任何异常 log 不退出。"""
    while True:
        try:
            await poll_tick(
                cfg=cfg, repo=repo, ledger=ledger, pg_pool=pg_pool,
                free_cpu_h=free_cpu_h,
                price_gpu=price_gpu, price_cpu=price_cpu,
            )
        except Exception:  # noqa: BLE001
            log.exception("poll_tick failed")
        await asyncio.sleep(interval_s)


async def poll_tick(
    *, cfg: SshConfig, repo: MdJobRepo, ledger: Ledger, pg_pool,
    free_cpu_h: float, price_gpu: int, price_cpu: int,
) -> None:
    """单次扫描:squeue 拉所有 active,离队列的进 sacct + finalize。"""
    active = await repo.list_active()
    if not active:
        return

    slurm_ids = [a["slurm_job_id"] for a in active if a["slurm_job_id"]]
    if not slurm_ids:
        return

    squeue_cmd = (
        f"squeue --noheader -j {','.join(slurm_ids)} "
        "-o '%i|%T|%R' 2>/dev/null || true"
    )
    squeue_out = await ssh_run(cfg, squeue_cmd)
    still_active = _parse_squeue(squeue_out)

    for job in active:
        sid = job["slurm_job_id"]
        if sid in still_active:
            new_status, node = still_active[sid]
            await _update_running_status(repo, job, new_status, node)
        else:
            await _finalize_via_sacct(
                cfg, repo, ledger, pg_pool, job,
                free_cpu_h, price_gpu, price_cpu,
            )


def _parse_squeue(out: str) -> dict[str, tuple[str, str]]:
    """{slurm_id: (state_lowercase, node)}"""
    result = {}
    for line in out.strip().splitlines():
        parts = line.split("|")
        if len(parts) < 3:
            continue
        sid, state, node = parts[0].strip(), parts[1].strip(), parts[2].strip()
        # squeue STATE: PENDING / RUNNING / ... → schema queued / running
        state_lc = state.lower()
        if state_lc == "pending":
            result[sid] = ("queued", node)
        elif state_lc == "running":
            result[sid] = ("running", node)
        else:
            # 其他状态(SUSPENDED / COMPLETING),保留作 running
            result[sid] = ("running", node)
    return result


async def _update_running_status(
    repo: MdJobRepo, job: dict, new_status: str, node: str,
) -> None:
    if job["status"] == new_status:
        # 已是 running,只更 gpu_host(若新拿到)
        if new_status == "running" and not job["gpu_host"] and node:
            await repo.update_status(job["id"], "running", gpu_host=node)
        return
    if new_status == "running":
        await repo.update_status(job["id"], "running", gpu_host=node or None)


async def _finalize_via_sacct(
    cfg: SshConfig, repo: MdJobRepo, ledger: Ledger, pg_pool,
    job: dict, free_cpu_h: float, price_gpu: int, price_cpu: int,
) -> None:
    sid = job["slurm_job_id"]
    sacct_cmd = (
        f"sacct -X -P --noheader -j {sid} "
        "-o JobID,State,CPUTimeRAW,Elapsed,AllocGRES"
    )
    try:
        sacct_out = await ssh_run(cfg, sacct_cmd)
    except Exception as e:  # noqa: BLE001
        log.warning("sacct failed for %s: %s", sid, e)
        return

    parsed = _parse_sacct(sacct_out, sid)
    if parsed is None:
        # sacct 拿不到 → 标 failed
        async with pg_pool.acquire() as conn:
            async with conn.transaction():
                await repo.finalize(
                    conn, job["id"], status="failed",
                    cpu_hours=0, gpu_hours=0, cost_li=0,
                    error=f"sacct: not_found ({sid})",
                )
        return

    state, cpu_seconds, elapsed_seconds, n_gpus = parsed
    cpu_h = cpu_seconds / 3600.0
    gpu_h = (elapsed_seconds / 3600.0) * n_gpus

    if state in _SACCT_OK:
        final_status = "done"
        cost = compute_md_cost_li(gpu_h, cpu_h, free_cpu_h, price_gpu, price_cpu)
        err = None
    elif state in _SACCT_CANCEL:
        final_status = "cancelled"
        cost = 0
        err = None
    elif state in _SACCT_FAIL:
        final_status = "failed"
        cost = 0
        err = f"slurm_{state.lower()}"
    else:
        final_status = "failed"
        cost = 0
        err = f"sacct_unknown_state:{state}"

    async with pg_pool.acquire() as conn:
        async with conn.transaction():
            await repo.finalize(
                conn, job["id"], status=final_status,
                cpu_hours=cpu_h, gpu_hours=gpu_h, cost_li=cost, error=err,
            )
            if cost > 0:
                await ledger.record(
                    conn, job["user_id"], -cost,
                    kind="md_consume", ref_type="md_job", ref_id=job["id"],
                    meta={"gpu_hours": gpu_h, "cpu_hours": cpu_h},
                )


def _parse_sacct(out: str, slurm_id: str) -> tuple[str, int, int, int] | None:
    """返 (state, cpu_seconds, elapsed_seconds, n_gpus). 找不到返 None。"""
    for line in out.strip().splitlines():
        parts = line.split("|")
        if len(parts) < 5:
            continue
        jobid, state, cpu_raw, elapsed, alloc_gres = parts[:5]
        if not jobid.startswith(slurm_id):
            continue
        return (
            state.strip(),
            int(cpu_raw or 0),
            _parse_elapsed(elapsed.strip()),
            _parse_gres_gpus(alloc_gres.strip()),
        )
    return None


def _parse_elapsed(s: str) -> int:
    """SLURM Elapsed 格式: 'HH:MM:SS' 或 'D-HH:MM:SS'. 返秒。"""
    if not s:
        return 0
    days = 0
    if "-" in s:
        d, s = s.split("-", 1)
        days = int(d)
    parts = s.split(":")
    if len(parts) != 3:
        return 0
    h, m, sec = (int(x) for x in parts)
    return days * 86400 + h * 3600 + m * 60 + sec


def _parse_gres_gpus(s: str) -> int:
    """AllocGRES 形如 'gpu:1' / 'gpu:rtx4090:2' / ''. 返 GPU 数,默认 1。"""
    if not s or s.lower() == "(null)":
        return 1
    if "gpu" in s.lower():
        # 取冒号后最后一个数字
        last = s.split(":")[-1]
        try:
            return max(1, int(last))
        except ValueError:
            return 1
    return 1
```

"""GPU 池状态查询 — 调 .53 上 ~/check_gpus.sh,解析 JSON。

from __future__ import annotations

import json
import logging
from typing import Any

from praxis2.md.ssh import SshConfig, ssh_run

log = logging.getLogger(__name__)


_EMPTY: dict[str, Any] = {
    "total_slots": 0, "busy_slots": 0, "free_slots": 0,

async def fetch_gpu_pool(*, cfg: SshConfig, script_path: str) -> dict[str, Any]:
    """跑 .53 上 check_gpus.sh,返 GpuPoolStatus dict。"""
    try:
        out = await ssh_run(cfg, f"bash {script_path}")
    except Exception:  # noqa: BLE001
        log.exception("gpu_pool ssh failed")
        return dict(_EMPTY)
    try:
        result = json.loads(out)
    except ValueError:
        log.warning("gpu_pool: malformed JSON: %r", out[:200])
        return dict(_EMPTY)
    # 防字段缺失
    for k, v in _EMPTY.items():
        result.setdefault(k, v)
    return result

```python
# praxis2/md/handlers.py
"""5 个 md ASGI handler。

挂在 server/app.py 的静态 _routes 和动态 dispatch:
- POST /api/md/submit                       handle_md_submit
- GET  /api/md/runs                         handle_md_runs
- GET  /api/md/jobs/{job_id}                handle_md_get
- POST /api/md/jobs/{job_id}/cancel         handle_md_cancel
- GET  /api/md/gpu-pool                     handle_md_gpu_pool
"""
from __future__ import annotations

import json
import logging
import uuid
from typing import Any

from praxis2.md.gpu_pool import fetch_gpu_pool
from praxis2.md.sbatch import render_sbatch
from praxis2.md.sim_params import validate_sim_params
from praxis2.md.ssh import ssh_run, ssh_write_file

log = logging.getLogger(__name__)


async def _json(send, status: int, body: dict) -> None:
    data = json.dumps(body, ensure_ascii=False, default=str).encode()
    await send({"type": "http.response.start", "status": status,
                "headers": [(b"content-type", b"application/json; charset=utf-8")]})
    await send({"type": "http.response.body", "body": data})


async def _read_body(receive) -> dict:
    chunks = b""
    while True:
        msg = await receive()
        chunks += msg.get("body", b"")
        if not msg.get("more_body"):
            break
    if not chunks:
        return {}
    try:
        return json.loads(chunks)
    except ValueError:
        return {}


def _require_user(scope) -> tuple[str, Any]:
    user = scope.get("user")
    if not user:
        return ("unauthorized", None)
    return ("ok", user)


async def _read_file_to_bytes(fs_store, user_id: str, file_id: str) -> bytes:
    """聚合 fs_store.read_stream 流为整 bytes。MD input 一般 <10MB,内存可承受。"""
    chunks: list[bytes] = []
    async for chunk in fs_store.read_stream(user_id, file_id):
        chunks.append(chunk)
    return b"".join(chunks)


_REQUIRED_TOP_FIELDS = [
    "structure_file_id", "force_field", "water_model", "ensemble",
    "temperature_k", "time_ns", "timestep_fs",
]


async def handle_md_submit(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return

    body = await _read_body(receive)
    sim_type = body.get("sim_type") or "standard"
    sim_params = body.get("sim_params") or {}

    # 顶级字段必填
    for f in _REQUIRED_TOP_FIELDS:
        if f not in body:
            await _json(send, 400, {"error": "missing_field", "field": f})
            return

    # sim_type 必填校验
    err = validate_sim_params(sim_type, sim_params)
    if err:
        # err 如 "missing: pull_rate" / "type: pull_rate" / "unsupported_sim_type"
        if err == "unsupported_sim_type":
            await _json(send, 400, {"error": "unsupported_sim_type",
                                    "sim_type": sim_type})
        elif err.startswith("missing: "):
            await _json(send, 400, {"error": "missing_sim_param",
                                    "field": err[len("missing: "):]})
        else:
            await _json(send, 400, {"error": "bad_sim_param",
                                    "field": err[len("type: "):]})
        return

    user_id = user["user_id"]
    repo = deps["md_repo"]

    # concurrent 限
    active = await repo.count_active(user_id)
    if active >= deps["max_concurrent_per_user"]:
        await _json(send, 429, {"error": "too_many_active",
                                "active": active,
                                "limit": deps["max_concurrent_per_user"]})
        return

    # quota 预检 — balance > 0
    async with deps["pg_pool"].acquire() as conn:
        bal = await conn.fetchval(
            "SELECT balance_li FROM users WHERE id=$1", uuid.UUID(user_id),
        )
    if bal is None or bal <= 0:
        await _json(send, 402, {"error": "insufficient_balance", "balance": bal})
        return

    # 拉 structure 文件(fs_store.read_stream 是 AsyncIterator)
    files_repo = deps["files_repo"]
    structure_fid = body["structure_file_id"]
    file_meta = await files_repo.get(structure_fid)
    if file_meta is None or str(file_meta["user_id"]) != user_id:
        await _json(send, 404, {"error": "structure_file_not_found"})
        return
    structure_bytes = await _read_file_to_bytes(
        deps["fs_store"], user_id, structure_fid,
    )

    plumed_bytes: bytes | None = None
    if sim_type == "metadynamics":
        pf_id = sim_params["plumed_file_id"]
        pf_meta = await files_repo.get(pf_id)
        if pf_meta is None or str(pf_meta["user_id"]) != user_id:
            await _json(send, 404, {"error": "plumed_file_not_found"})
            return
        plumed_bytes = await _read_file_to_bytes(
            deps["fs_store"], user_id, pf_id,
        )

    # 组 params(含 sim_params)
    params = {f: body[f] for f in _REQUIRED_TOP_FIELDS}
    params["notes"] = body.get("notes", "")
    params["sim_params"] = sim_params
    params["structure_file_id"] = structure_fid

    # 创 PG row
    async with deps["pg_pool"].acquire() as conn:
        job_id = await repo.create(conn, user_id, sim_type, params)

    # ssh 走全套 stage + sbatch
    cfg = deps["ssh_cfg"]
    work_dir = f"~/praxis-md/{job_id}"
    account = deps["hpc_account"]

    try:
        await ssh_run(cfg, f"mkdir -p {work_dir}")
        await ssh_write_file(cfg, f"{work_dir}/input.pdb", structure_bytes)
        if plumed_bytes is not None:
            await ssh_write_file(cfg, f"{work_dir}/input.plumed", plumed_bytes)
        script = render_sbatch(
            sim_type=sim_type, params=params, sim_params=sim_params,
            work_dir=work_dir, account=account,
        )
        await ssh_write_file(cfg, f"{work_dir}/job.sh", script.encode())
        sbatch_out = await ssh_run(cfg, f"sbatch {work_dir}/job.sh")
    except Exception as exc:  # noqa: BLE001
        log.exception("md submit ssh chain failed")
        await repo.update_cancelled(job_id)
        await _json(send, 502, {"error": "hpc_unreachable", "reason": repr(exc)[:300]})
        return

    # parse "Submitted batch job 12345"
    parts = sbatch_out.strip().rsplit(maxsplit=1)
    if len(parts) < 2 or not parts[-1].isdigit():
        log.error("sbatch output unparseable: %r", sbatch_out)
        await repo.update_cancelled(job_id)
        await _json(send, 502, {"error": "sbatch_unparseable",
                                "stdout": sbatch_out[:300]})
        return
    slurm_job_id = parts[-1]
    await repo.update_submitted(job_id, slurm_job_id=slurm_job_id)

    await _json(send, 200, {"id": job_id, "status": "queued",
                            "slurm_job_id": slurm_job_id})


async def handle_md_runs(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return
    rows = await deps["md_repo"].list_recent(user["user_id"], limit=100)
    await _json(send, 200, {"runs": rows})


async def handle_md_get(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return
    job_id = scope.get("path_params", {}).get("job_id", "")
    if not job_id:
        await _json(send, 400, {"error": "missing_job_id"})
        return
    row = await deps["md_repo"].get(job_id)
    if row is None:
        await _json(send, 404, {"error": "not_found"})
        return
    if row["user_id"] != user["user_id"] and user.get("role") != "admin":
        await _json(send, 404, {"error": "not_found"})   # 不泄露存在
        return
    await _json(send, 200, row)


async def handle_md_cancel(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return
    job_id = scope.get("path_params", {}).get("job_id", "")
    if not job_id:
        await _json(send, 400, {"error": "missing_job_id"})
        return
    repo = deps["md_repo"]
    row = await repo.get(job_id)
    if row is None or (row["user_id"] != user["user_id"]
                       and user.get("role") != "admin"):
        await _json(send, 404, {"error": "not_found"})
        return
    if row["status"] not in ("queued", "running"):
        await _json(send, 409, {"error": "not_active", "status": row["status"]})
        return

    # 调 scancel(若 slurm_job_id 已写)
    if row["slurm_job_id"]:
        try:
            await ssh_run(deps["ssh_cfg"], f"scancel {row['slurm_job_id']}")
        except Exception as exc:  # noqa: BLE001
            log.warning("scancel ssh failed for %s: %s", row["slurm_job_id"], exc)
            # 仍把 DB 标 cancelled — 后台 poll worker 下次会经 sacct 确认
    await repo.update_cancelled(job_id)
    await _json(send, 200, {"ok": True})


async def handle_md_gpu_pool(scope, receive, send, deps) -> None:
    status, user = _require_user(scope)
    if status != "ok":
        await _json(send, 401, {"error": "unauthorized"})
        return
    pool = await fetch_gpu_pool(
        cfg=deps["ssh_cfg"], script_path=deps["check_gpus_script_path"],
    )
    await _json(send, 200, pool)

#!/usr/bin/env bash
# 顺序 ssh 到 GPU 节点拉 nvidia-smi,聚合为 JSON。
# 若 .53 装了 pdsh,可改 parallel 版,但顺序版无外部依赖更稳。
#
# 部署:scp scripts/check_gpus.sh user@10.102.106.53:~/check_gpus.sh && chmod +x
# 环境覆盖 GPU 节点列表:export PRAXIS_GPU_NODES="bio000071 bio000072 bio000073"
set -euo pipefail
NODES=(${PRAXIS_GPU_NODES:-bio000071 bio000072 bio000073 bio000074})
python3 - "${NODES[@]}" <<'PY'
import json, subprocess, sys
nodes = sys.argv[1:]
gpus = []
for h in nodes:
    try:
        out = subprocess.check_output(
            ["ssh", "-o", "ConnectTimeout=5", "-o", "BatchMode=yes", h,
             "nvidia-smi --query-gpu=index,name,memory.used,memory.total,"
             "utilization.gpu --format=csv,noheader,nounits"],
            text=True, timeout=15,
        )
    except subprocess.SubprocessError:
        for i in range(8):
            gpus.append({
                "host": h, "gpu_id": i, "name": "?",
                "mem_mb": "0/0", "util_pct": 0,
                "busy": True, "busy_label": "unreachable",
            })
        continue
    for line in out.strip().splitlines():
        idx, name, mem_used, mem_total, util = [s.strip() for s in line.split(",")]
        busy = int(util) > 5 or int(mem_used) > 500
        gpus.append({
            "host": h, "gpu_id": int(idx), "name": name,
            "mem_mb": f"{mem_used}/{mem_total}",
            "util_pct": int(util),
            "busy": busy, "busy_label": None if not busy else "in_use",
        })
busy_n = sum(1 for g in gpus if g["busy"])
print(json.dumps({
    "total_slots": len(gpus),
    "busy_slots": busy_n,
    "free_slots": len(gpus) - busy_n,
    "hosts": nodes,
    "gpus": gpus,
}))
PY
```
